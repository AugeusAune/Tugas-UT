A. Analisis Skema dan Relasi ERD

ERD kasus rumah sakit terdiri dari enam entitas utama. Setiap entitas dimodelkan ke dalam tabel basis data relasional dengan atribut kunci utama bertipe integer auto-increment serta relasi referensial antar tabel menggunakan kunci tamu (foreign key).

Langkah 3Pembuatan Basis Data dari Command Prompt / PhpMyAdmin

Pembuatan basis data utama praktikum dengan nama rumah_sakit menggunakan sintaks DDL:

CREATE DATABASE IF NOT EXISTS rumah_sakit;
SHOW DATABASES;
USE rumah_sakit;

Langkah 4Drop Database

Perintah DROP DATABASE menghapus basis data beserta seluruh tabel dan rekaman data di dalamnya secara permanen. Pengujian dilakukan dengan membuat basis data sementara kemudian menghapusnya:

-- Membuat database pengujian sementara:
CREATE DATABASE IF NOT EXISTS db_temp_praktikum;
SHOW DATABASES;

-- Menghapus database pengujian sementara:
DROP DATABASE db_temp_praktikum;
SHOW DATABASES;

-- Kembali aktif ke basis data tugas utama:
USE rumah_sakit;

Setelah dieksekusi, basis data db_temp_praktikum berhasil dihapus dan tidak lagi muncul pada daftar SHOW DATABASES.

Langkah 5Pembuatan Seluruh Tabel dari ERD

Seluruh tabel dibuat berurutan dari tabel master induk tanpa foreign key, kemudian tabel transaksi anak yang memiliki foreign key.

1) Tabel administrator

CREATE TABLE IF NOT EXISTS administrator (
    id_admin INT(3) PRIMARY KEY AUTO_INCREMENT,
    nama_admin VARCHAR(30) NOT NULL,
    waktu_jaga VARCHAR(100) NOT NULL
) ENGINE=InnoDB;

2) Tabel dokter

CREATE TABLE IF NOT EXISTS dokter (
    id_dokter INT(3) PRIMARY KEY AUTO_INCREMENT,
    nama_dokter VARCHAR(30) NOT NULL,
    alamat_dokter VARCHAR(100) NOT NULL,
    tanggal_lahir DATE NOT NULL,
    no_hp VARCHAR(15) NOT NULL,
    spesialis VARCHAR(30) NOT NULL,
    waktu_kerja VARCHAR(100) NOT NULL
) ENGINE=InnoDB;

3) Tabel pasien

CREATE TABLE IF NOT EXISTS pasien (
    id_pasien INT(3) PRIMARY KEY AUTO_INCREMENT,
    nama_pasien VARCHAR(30) NOT NULL,
    alamat_pasien VARCHAR(100) NOT NULL,
    jenis_kelamin INT(1) NOT NULL
) ENGINE=InnoDB;

4) Tabel dokter_admin

CREATE TABLE IF NOT EXISTS dokter_admin (
    id_data INT(3) PRIMARY KEY AUTO_INCREMENT,
    id_dokter INT(3) NOT NULL,
    id_admin INT(3) NOT NULL
) ENGINE=InnoDB;

5) Tabel daftar

CREATE TABLE IF NOT EXISTS daftar (
    id_daftar INT(3) PRIMARY KEY AUTO_INCREMENT,
    id_pasien INT(3) NOT NULL,
    id_admin INT(3) NOT NULL
) ENGINE=InnoDB;

6) Tabel pasien_dokter

CREATE TABLE IF NOT EXISTS pasien_dokter (
    id INT(3) PRIMARY KEY AUTO_INCREMENT,
    id_dokter INT(3) NOT NULL,
    id_pasien INT(3) NOT NULL,
    waktu_periksa DATE NOT NULL,
    resep VARCHAR(100) NOT NULL
) ENGINE=InnoDB;

Verifikasi seluruh tabel yang telah terbentuk:

SHOW TABLES;

Langkah 6Setting Primary Key, Foreign Key, dan Relasi Tabel

Kunci utama (Primary Key) telah didefinisikan saat tabel dibuat. Relasi antar tabel diatur menggunakan perintah ALTER TABLE dengan penegakan integritas referensial ON DELETE CASCADE ON UPDATE CASCADE:

-- 1. Relasi tabel dokter_admin
ALTER TABLE dokter_admin
    ADD CONSTRAINT fk_dokter_admin_dokter FOREIGN KEY (id_dokter) REFERENCES dokter(id_dokter) ON DELETE CASCADE ON UPDATE CASCADE,
    ADD CONSTRAINT fk_dokter_admin_admin FOREIGN KEY (id_admin) REFERENCES administrator(id_admin) ON DELETE CASCADE ON UPDATE CASCADE;

-- 2. Relasi tabel daftar
ALTER TABLE daftar
    ADD CONSTRAINT fk_daftar_pasien FOREIGN KEY (id_pasien) REFERENCES pasien(id_pasien) ON DELETE CASCADE ON UPDATE CASCADE,
    ADD CONSTRAINT fk_daftar_admin FOREIGN KEY (id_admin) REFERENCES administrator(id_admin) ON DELETE CASCADE ON UPDATE CASCADE;

-- 3. Relasi tabel pasien_dokter
ALTER TABLE pasien_dokter
    ADD CONSTRAINT fk_pasien_dokter_dokter FOREIGN KEY (id_dokter) REFERENCES dokter(id_dokter) ON DELETE CASCADE ON UPDATE CASCADE,
    ADD CONSTRAINT fk_pasien_dokter_pasien FOREIGN KEY (id_pasien) REFERENCES pasien(id_pasien) ON DELETE CASCADE ON UPDATE CASCADE;

Verifikasi foreign key aktif pada katalog sistem information_schema:

SELECT TABLE_NAME, COLUMN_NAME, CONSTRAINT_NAME, REFERENCED_TABLE_NAME, REFERENCED_COLUMN_NAME
FROM information_schema.KEY_COLUMN_USAGE
WHERE TABLE_SCHEMA = 'rumah_sakit' AND REFERENCED_TABLE_NAME IS NOT NULL;

Langkah 8Penggunaan SELECT

Perintah SELECT mengambil data seluruh kolom maupun kolom spesifik dengan alias (AS):

-- Menampilkan seluruh kolom tabel dokter:
SELECT * FROM dokter;

-- Menampilkan kolom terpilih dari tabel pasien dengan alias:
SELECT 
    id_pasien AS 'ID Pasien', 
    nama_pasien AS 'Nama Lengkap Pasien', 
    alamat_pasien AS 'Alamat Domisili'
FROM pasien;

Langkah 9Penggunaan WHERE

Klausa WHERE memfilter data berdasarkan kondisi tertentu:

-- Dokter spesialis Penyakit Dalam:
SELECT id_dokter, nama_dokter, spesialis, no_hp 
FROM dokter 
WHERE spesialis = 'Penyakit Dalam';

-- Pasien berjenis kelamin perempuan (2):
SELECT id_pasien, nama_pasien, alamat_pasien 
FROM pasien 
WHERE jenis_kelamin = 2;

-- Pemeriksaan oleh dokter 1 setelah tanggal 12 Oktober:
SELECT id, id_dokter, id_pasien, waktu_periksa, resep
FROM pasien_dokter
WHERE id_dokter = 1 AND waktu_periksa >= '2026-10-12';

Langkah 10Penggunaan JOIN

Operasi JOIN menggabungkan data antar tabel berdasarkan relasi foreign key:

-- INNER JOIN 3 tabel (Pasien, Riwayat Pemeriksaan, Dokter):
SELECT 
    pd.id AS no_pemeriksaan,
    p.nama_pasien,
    d.nama_dokter,
    d.spesialis,
    pd.waktu_periksa,
    pd.resep
FROM pasien_dokter pd
JOIN pasien p ON pd.id_pasien = p.id_pasien
JOIN dokter d ON pd.id_dokter = d.id_dokter;

Langkah 11Penggunaan Fungsi Matematika

Menggunakan fungsi matematika dan agregasi bawaan (COUNT, AVG, ROUND, MIN, MAX, TIMESTAMPDIFF):

-- Menghitung total data:
SELECT 
    (SELECT COUNT(*) FROM dokter) AS total_dokter,
    (SELECT COUNT(*) FROM pasien) AS total_pasien,
    (SELECT COUNT(*) FROM administrator) AS total_admin,
    (SELECT COUNT(*) FROM pasien_dokter) AS total_pemeriksaan;

-- Menghitung rata-rata usia dokter dan usia termuda/tertua:
SELECT 
    ROUND(AVG(TIMESTAMPDIFF(YEAR, tanggal_lahir, '2026-11-01')), 1) AS rata_rata_usia_dokter,
    MIN(tanggal_lahir) AS tgl_lahir_dokter_tertua,
    MAX(tanggal_lahir) AS tgl_lahir_dokter_termuda
FROM dokter;

Langkah 12Penggunaan Fungsi String

Fungsi string (CONCAT, UPPER, LOWER, LENGTH, SUBSTRING) memanipulasi teks:

-- Menggabungkan gelar dan spesialisasi dokter:
SELECT CONCAT('dr. ', nama_dokter, ' (Spesialis ', spesialis, ')') AS profil_dokter FROM dokter;

-- Standarisasi teks kapital dan kecil:
SELECT UPPER(nama_pasien) AS nama_kapital, LOWER(alamat_pasien) AS alamat_huruf_kecil FROM pasien;

-- Menghitung panjang karakter dan mengekstrak kode resep:
SELECT nama_pasien, LENGTH(nama_pasien) AS panjang_karakter FROM pasien;
SELECT id, resep, SUBSTRING(resep, 1, 7) AS kode_resep, SUBSTRING(resep, 9) AS rincian_obat FROM pasien_dokter;

Langkah 13Penggunaan BETWEEN

Operator BETWEEN menyeleksi data dalam interval rentang nilai:

-- Dokter dengan rentang tanggal lahir tahun 1980 sampai 1989:
SELECT nama_dokter, spesialis, tanggal_lahir 
FROM dokter 
WHERE tanggal_lahir BETWEEN '1980-01-01' AND '1989-12-31';

-- Pemeriksaan dalam rentang tanggal 10 Oktober s.d. 18 Oktober 2026:
SELECT id, id_pasien, waktu_periksa, resep 
FROM pasien_dokter 
WHERE waktu_periksa BETWEEN '2026-10-10' AND '2026-10-18';

Langkah 14Penggunaan LIKE

Pencarian pola teks dengan wildcard %:

-- Nama pasien berawalan huruf 'S':
SELECT id_pasien, nama_pasien, alamat_pasien FROM pasien WHERE nama_pasien LIKE 'S%';

-- Nama dokter berakhiran kata 'Wijaya':
SELECT id_dokter, nama_dokter, spesialis FROM dokter WHERE nama_dokter LIKE '%Wijaya';

-- Dokter dengan alamat domisili di Jakarta:
SELECT id_dokter, nama_dokter, alamat_dokter FROM dokter WHERE alamat_dokter LIKE '%Jakarta%';

Langkah 15Penggunaan ORDER BY

Pengurutan data secara menaik (ASC) atau menurun (DESC):

-- Ascending (A-Z) berdasarkan nama pasien:
SELECT * FROM pasien ORDER BY nama_pasien ASC;

-- Descending (termuda ke tertua) berdasarkan tanggal lahir dokter:
SELECT nama_dokter, tanggal_lahir, spesialis FROM dokter ORDER BY tanggal_lahir DESC;

-- Multi-kolom:
SELECT id, waktu_periksa, id_dokter, id_pasien, resep FROM pasien_dokter ORDER BY waktu_periksa DESC, id_dokter ASC;

Langkah 16Penggunaan GROUP BY

Pengelompokan data dengan agregasi dan klausa HAVING:

-- Jumlah pasien berdasarkan jenis kelamin:
SELECT 
    CASE jenis_kelamin WHEN 1 THEN 'Laki-laki' ELSE 'Perempuan' END AS jenis_kelamin,
    COUNT(*) AS total_pasien
FROM pasien 
GROUP BY jenis_kelamin;

-- Jumlah pemeriksaan per dokter dengan HAVING:
SELECT d.nama_dokter, COUNT(pd.id) AS total_pemeriksaan
FROM dokter d
JOIN pasien_dokter pd ON d.id_dokter = pd.id_dokter
GROUP BY d.id_dokter, d.nama_dokter
HAVING COUNT(pd.id) >= 2;

Langkah 17Penggunaan UPDATE

Pembaruan data dan verifikasi hasil perubahannya:

-- Memperbarui kontak dan jam kerja dokter spesialis (id_dokter = 1):
UPDATE dokter 
SET no_hp = '081299998888', waktu_kerja = 'Senin-Jumat 08.00-15.00' 
WHERE id_dokter = 1;
SELECT id_dokter, nama_dokter, no_hp, waktu_kerja FROM dokter WHERE id_dokter = 1;

-- Memperbarui domisili pasien (id_pasien = 1):
UPDATE pasien 
SET alamat_pasien = 'Jl. Boulevard Raya No. 88 Bekasi Barat' 
WHERE id_pasien = 1;
SELECT id_pasien, nama_pasien, alamat_pasien FROM pasien WHERE id_pasien = 1;

-- Memperbarui resep obat (id = 1):
UPDATE pasien_dokter 
SET resep = 'RSP-001 Paracetamol 500mg, Amoxicillin 500mg, Vitamin B Complex' 
WHERE id = 1;
SELECT id, id_pasien, waktu_periksa, resep FROM pasien_dokter WHERE id = 1;
